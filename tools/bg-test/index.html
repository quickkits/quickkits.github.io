// Sample JavaScript code
const theme = {
  background: "#EDE3D3",
  foreground: "#1F1A14",
  fontSize: 16
};

function applyTheme(theme) {
  document.body.style.backgroundColor = theme.background;
  document.body.style.color = theme.foreground;
  console.log("Theme applied successfully");
}

applyTheme(theme);
